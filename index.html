// PULSE — Home Page // Controls index.html only. document.addEventListener("DOMContentLoaded", () => { const logoutBtn = document.getElementById("logoutBtn"); if (!logoutBtn) { return; } logoutBtn.addEventListener("click", async () => { logoutBtn.disabled = true; logoutBtn.textContent = "Signing Out..."; try { const success = await signOut(); if (!success) { logoutBtn.disabled = false; logoutBtn.textContent = "Sign Out"; alert("Unable to sign out. Please try again."); } } catch (error) { console.error("Home sign-out error:", error); logoutBtn.disabled = false; logoutBtn.textContent = "Sign Out"; alert("Unable to sign out. Please try again."); } }); });
