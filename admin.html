// PULSE — Admin Dashboard // Controls admin.html only. document.addEventListener("DOMContentLoaded", () => { loadAdminDashboard(); }); async function loadAdminDashboard() { const container = document.getElementById("adminContent"); if (!container) { console.error( "PULSE: adminContent element not found." ); return; } container.innerHTML = "Loading admin dashboard..."; try { // ============================== // CHECK SUPABASE CLIENT // ============================== if (!window.supabaseClient) { throw new Error( "Supabase client is not available." ); } // ============================== // GET CURRENT USER // ============================== const user = await getCurrentUser(); if (!user) { container.innerHTML = `

You must be logged in to access the Admin Dashboard.

Login `; return; } // ============================== // VERIFY ADMIN // ============================== const admin = await isAdmin(user.id); if (!admin) { container.innerHTML = `

Access denied.

This page is available to administrators only.

Return to Dashboard `; return; } console.log( "PULSE: ADMIN DASHBOARD ACCESS GRANTED" ); // ============================== // LOAD ADMIN COUNTS // ============================== const [ tasksResult, submissionsResult, businessesResult, rewardsResult ] = await Promise.all([ window.supabaseClient .from("tasks") .select("id", { count: "exact", head: true }), window.supabaseClient .from("task_submissions") .select("id", { count: "exact", head: true }), window.supabaseClient .from("businesses") .select("id", { count: "exact", head: true }), window.supabaseClient .from("redemption_requests") .select("id", { count: "exact", head: true }) ]); // ============================== // CHECK QUERY ERRORS // ============================== if (tasksResult.error) { throw tasksResult.error; } if (submissionsResult.error) { throw submissionsResult.error; } if (businessesResult.error) { throw businessesResult.error; } if (rewardsResult.error) { throw rewardsResult.error; } // ============================== // DISPLAY ADMIN DASHBOARD // ============================== container.innerHTML = `

Tasks

${tasksResult.count ?? 0}

Manage Tasks

Submissions

${submissionsResult.count ?? 0}

Review Submissions

Businesses

${businessesResult.count ?? 0}

Manage Businesses

Reward Requests

${rewardsResult.count ?? 0}

Review Rewards

Reports & Records

View system records and reports.

Open Reports
`; } catch (error) { console.error( "PULSE: Admin dashboard error:", error ); container.innerHTML = `

Unable to load the Admin Dashboard.

Please try again.

`; } }
